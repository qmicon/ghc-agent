<script>
document.addEventListener('DOMContentLoaded', function() {
  const faqItems = document.querySelectorAll('.offer-faq__item');
  
  faqItems.forEach(item => {
    const question = item.querySelector('.offer-faq__question');
    const answer = item.querySelector('.offer-faq__answer');
    const answerInner = item.querySelector('.offer-faq__answer-inner');
    
    question.addEventListener('click', () => {
      const expanded = question.getAttribute('aria-expanded') === 'true';
      
      // Close all other FAQs
      faqItems.forEach(otherItem => {
        if (otherItem !== item) {
          const otherQuestion = otherItem.querySelector('.offer-faq__question');
          const otherAnswer = otherItem.querySelector('.offer-faq__answer');
          
          otherQuestion.setAttribute('aria-expanded', 'false');
          otherAnswer.style.height = '0';
        }
      });
      
      // Toggle current FAQ
      question.setAttribute('aria-expanded', !expanded);
      
      if (!expanded) {
        answer.style.height = answerInner.offsetHeight + 'px';
      } else {
        answer.style.height = '0';
      }
    });
  });
});
</script>

document.addEventListener('DOMContentLoaded', function() {
  const track = document.getElementById('testimonial-track');
  const slides = track ? track.querySelectorAll('.testimonial-slide') : [];
  const prevButton = document.querySelector('.testimonial-prev');
  const nextButton = document.querySelector('.testimonial-next');
  
  if (!track || slides.length <= 1) return;
  
  let currentIndex = 0;
  let autoplayInterval;
  
  // Initialize slider
  function initializeSlider() {
    updateSlides();
    startAutoplay();
  }
  
  // Update slide positions and active states
  function updateSlides() {
    // Position all slides with transform
    track.style.transform = `translateX(-${currentIndex * 100}%)`;
    
    // Update active classes for animation
    slides.forEach((slide, index) => {
      slide.classList.toggle('active', index === currentIndex);
    });
  }
  
  // Navigation functions
  function goToSlide(index) {
    currentIndex = (index + slides.length) % slides.length;
    updateSlides();
    restartAutoplay();
  }
  
  function goToPrev() {
    goToSlide(currentIndex - 1);
  }
  
  function goToNext() {
    goToSlide(currentIndex + 1);
  }
  
  // Autoplay functions
  function startAutoplay() {
    autoplayInterval = setInterval(goToNext, 5000);
  }
  
  function restartAutoplay() {
    clearInterval(autoplayInterval);
    startAutoplay();
  }
  
  // Event listeners
  if (prevButton) {
    prevButton.addEventListener('click', goToPrev);
  }
  
  if (nextButton) {
    nextButton.addEventListener('click', goToNext);
  }
  
  // Initialize the slider
  initializeSlider();
});
</script>

<script>
document.addEventListener('DOMContentLoaded', function() {
  const countdownElements = document.querySelectorAll('.offer-countdown');
  
  countdownElements.forEach(countdown => {
    const endDateStr = countdown.getAttribute('data-end-date');
    if (!endDateStr) return;
    
    const endDate = new Date(endDateStr).getTime();
    const daysEl = countdown.querySelector('.offer-countdown__days');
    const hoursEl = countdown.querySelector('.offer-countdown__hours');
    const minutesEl = countdown.querySelector('.offer-countdown__minutes');
    const secondsEl = countdown.querySelector('.offer-countdown__seconds');
    const expiredEl = countdown.querySelector('.offer-countdown__expired');
    const containerEl = countdown.querySelector('.offer-countdown__container');
    
    function updateCountdown() {
      const now = new Date().getTime();
      const distance = endDate - now;
      
      if (distance < 0) {
        // Countdown expired
        clearInterval(interval);
        containerEl.style.display = 'none';
        expiredEl.style.display = 'block';
        return;
      }
      
      // Calculate time units
      const days = Math.floor(distance / (1000 * 60 * 60 * 24));
      const hours = Math.floor((distance % (1000 * 60 * 60 * 24)) / (1000 * 60 * 60));
      const minutes = Math.floor((distance % (1000 * 60 * 60)) / (1000 * 60));
      const seconds = Math.floor((distance % (1000 * 60)) / 1000);
      
      // Update elements with leading zeros
      daysEl.textContent = days.toString().padStart(2, '0');
      hoursEl.textContent = hours.toString().padStart(2, '0');
      minutesEl.textContent = minutes.toString().padStart(2, '0');
      secondsEl.textContent = seconds.toString().padStart(2, '0');
    }
    
    // Initial update
    updateCountdown();
    
    // Update every second
    const interval = setInterval(updateCountdown, 1000);
  });
});
</script>
