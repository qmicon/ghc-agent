<script>
  class OfferProductGrid {
    constructor() {
      this.selectedProducts = [];
      this.maxSelection = 3;
      this.init();
    }
    
    init() {
      // Update counter
      this.updateCounter();
      
      // Initialize product cards
      this.initProductCards();
      
      // Listen for slot update events
      document.addEventListener('offer:slot-updated', (event) => {
        const { action, productId, totalSelected } = event.detail;
        if (action === 'remove') {
          this.removeProduct(productId);
        }
        this.updateCounter();
      });
    }
    
    initProductCards() {
      const productCards = document.querySelectorAll('.offer-product-card');
      
      productCards.forEach(card => {
        const productId = card.dataset.productId;
        const selectButton = card.querySelector('.offer-product-card__select-button');
        
        // Check if product is already selected
        if (this.isProductSelected(productId)) {
          this.markCardAsSelected(card);
        }
        
        selectButton.addEventListener('click', () => {
          if (this.isProductSelected(productId)) {
            this.removeProduct(productId);
            this.markCardAsUnselected(card);
          } else {
            if (this.selectedProducts.length < this.maxSelection) {
              this.addProduct({
                id: productId,
                title: card.dataset.productTitle,
                image: card.dataset.productImage,
                url: card.dataset.productUrl,
                variantId: card.dataset.variantId
              });
              this.markCardAsSelected(card);
            } else {
              this.showMaxSelectionAlert();
            }
          }
        });
      });
    }
    
    isProductSelected(productId) {
      return this.selectedProducts.some(product => product.id === productId);
    }
    
    markCardAsSelected(card) {
      card.classList.add('offer-product-card--selected');
      const button = card.querySelector('.offer-product-card__select-button');
      button.textContent = 'Selected';
      button.classList.add('offer-product-card__select-button--selected');
    }
    
    markCardAsUnselected(card) {
      card.classList.remove('offer-product-card--selected');
      const button = card.querySelector('.offer-product-card__select-button');
      button.textContent = 'Select';
      button.classList.remove('offer-product-card__select-button--selected');
    }
    
    addProduct(product) {
      if (this.selectedProducts.length < this.maxSelection) {
        this.selectedProducts.push(product);
        this.updateCounter();
        this.notifyProductAdded(product);
      }
    }
    
    removeProduct(productId) {
      const index = this.selectedProducts.findIndex(product => product.id === productId);
      if (index !== -1) {
        const removedProduct = this.selectedProducts.splice(index, 1)[0];
        this.updateCounter();
        
        // Update UI for the product card if it exists on the page
        const card = document.querySelector(`.offer-product-card[data-product-id="${productId}"]`);
        if (card) {
          this.markCardAsUnselected(card);
        }

        // Notify slots section about the removal
        document.dispatchEvent(new CustomEvent('offer:product-removed', {
          detail: { productId }
        }));
      }
    }
    
    updateCounter() {
      const counter = document.getElementById('selected-count');
      if (counter) {
        counter.textContent = this.selectedProducts.length;
      }
    }
    
    notifyProductAdded(product) {
      document.dispatchEvent(new CustomEvent('offer:product-added', {
        detail: { product }
      }));
    }
    
    showMaxSelectionAlert() {
      // Create a floating alert
      const alert = document.createElement('div');
      alert.className = 'offer-max-selection-alert';
      alert.textContent = 'You can only select 3 products. Remove a product to add a different one.';
      
      document.body.appendChild(alert);
      
      // Remove after 3 seconds
      setTimeout(() => {
        alert.classList.add('offer-max-selection-alert--hide');
        setTimeout(() => {
          document.body.removeChild(alert);
        }, 300);
      }, 3000);
    }
  }
  
  // Initialize on DOM content loaded
  document.addEventListener('DOMContentLoaded', () => {
    new OfferProductGrid();
  });
</script>

<script>
  class OfferSelectedProducts {
    constructor() {
      this.selectedProducts = [];
      this.maxSelection = 3;
      this.init();
    }
    
    init() {
      // Initialize slots
      this.updateSlots();
      
      // Initialize checkout button
      this.initCheckoutButton();
      
      // Listen for product selection events
      document.addEventListener('offer:product-added', (event) => {
        const { product } = event.detail;
        this.addProduct(product);
      });
      
      document.addEventListener('offer:product-removed', (event) => {
        const { productId } = event.detail;
        this.removeProduct(productId);
      });
      
      // Initialize remove buttons in slots
      this.initRemoveButtons();
    }
    
    updateSlots() {
      for (let i = 0; i < this.maxSelection; i++) {
        const slot = document.getElementById(`offer-slot-${i + 1}`);
        if (!slot) continue;
        
        const product = this.selectedProducts[i];
        
        if (product) {
          this.fillSlot(slot, product);
        } else {
          this.emptySlot(slot);
        }
      }
      
      // Update checkout button state
      this.updateCheckoutButton();
    }
    
    fillSlot(slot, product) {
      const slotContent = slot.querySelector('.offer-selected-product-slot__content');
      const slotImage = slot.querySelector('.offer-selected-product-slot__image');
      const slotTitle = slot.querySelector('.offer-selected-product-slot__title');
      const slotEmpty = slot.querySelector('.offer-selected-product-slot__empty');
      const removeButton = slot.querySelector('.offer-selected-product-slot__remove');
      
      // Update slot with product info
      if (slotImage) {
        if (product.image) {
          slotImage.innerHTML = `<img src="${product.image}" alt="${product.title}">`;
        } else {
          slotImage.innerHTML = '<div class="offer-selected-product-slot__no-image"></div>';
        }
      }
      
      if (slotTitle) {
        slotTitle.textContent = product.title;
      }
      
      // Show content, hide empty state
      if (slotContent) slotContent.style.display = 'flex';
      if (slotEmpty) slotEmpty.style.display = 'none';
      if (removeButton) {
        removeButton.style.display = 'block';
        removeButton.dataset.productId = product.id;
      }
      
      // Mark slot as filled
      slot.classList.add('offer-selected-product-slot--filled');
      slot.dataset.productId = product.id;
    }
    
    emptySlot(slot) {
      const slotContent = slot.querySelector('.offer-selected-product-slot__content');
      const slotEmpty = slot.querySelector('.offer-selected-product-slot__empty');
      const removeButton = slot.querySelector('.offer-selected-product-slot__remove');
      
      // Hide content, show empty state
      if (slotContent) slotContent.style.display = 'none';
      if (slotEmpty) slotEmpty.style.display = 'flex';
      if (removeButton) removeButton.style.display = 'none';
      
      // Mark slot as empty
      slot.classList.remove('offer-selected-product-slot--filled');
      slot.dataset.productId = '';
    }
    
    addProduct(product) {
      // Check if product is already in the list
      const existingIndex = this.selectedProducts.findIndex(p => p.id === product.id);
      if (existingIndex !== -1) return;
      
      // Add product if we haven't reached the max
      if (this.selectedProducts.length < this.maxSelection) {
        this.selectedProducts.push(product);
        this.updateSlots();
        
        // Dispatch event to update product selector
        document.dispatchEvent(new CustomEvent('offer:slot-updated', {
          detail: { 
            action: 'add',
            productId: product.id,
            totalSelected: this.selectedProducts.length
          }
        }));
      }
    }
    
    removeProduct(productId) {
      const index = this.selectedProducts.findIndex(product => product.id === productId);
      if (index !== -1) {
        this.selectedProducts.splice(index, 1);
        this.updateSlots();
        
        // Dispatch event to update product selector
        document.dispatchEvent(new CustomEvent('offer:slot-updated', {
          detail: { 
            action: 'remove',
            productId: productId,
            totalSelected: this.selectedProducts.length
          }
        }));
      }
    }
    
    initRemoveButtons() {
      document.querySelectorAll('.offer-selected-product-slot__remove').forEach(button => {
        button.addEventListener('click', (event) => {
          const productId = event.currentTarget.dataset.productId;
          if (productId) {
            this.removeProduct(productId);
          }
        });
      });
    }
    
    initCheckoutButton() {
      const checkoutButton = document.getElementById('offer-checkout-button');
      if (!checkoutButton) return;
      
      checkoutButton.addEventListener('click', () => {
        if (this.selectedProducts.length === this.maxSelection) {
          this.proceedToCheckout();
        }
      });
    }
    
    updateCheckoutButton() {
      const checkoutButton = document.getElementById('offer-checkout-button');
      if (!checkoutButton) return;
      
      if (this.selectedProducts.length === this.maxSelection) {
        checkoutButton.disabled = false;
      } else {
        checkoutButton.disabled = true;
      }
    }
    
    proceedToCheckout() {
      // Create a form to submit the selected products to cart
      const form = document.createElement('form');
      form.method = 'post';
      form.action = '/cart/add';
      
      // Add the selected products as line items
      this.selectedProducts.forEach((product, index) => {
        const variantInput = document.createElement('input');
        variantInput.type = 'hidden';
        variantInput.name = `items[${index}][id]`;
        variantInput.value = product.variantId;
        form.appendChild(variantInput);
        
        const quantityInput = document.createElement('input');
        quantityInput.type = 'hidden';
        quantityInput.name = `items[${index}][quantity]`;
        quantityInput.value = '1';
        form.appendChild(quantityInput);
        
        // Add a property to identify this as part of the offer
        const propertyInput = document.createElement('input');
        propertyInput.type = 'hidden';
        propertyInput.name = `items[${index}][properties][_offer]`;
        propertyInput.value = 'Buy 3 for ₹99';
        form.appendChild(propertyInput);
      });
      
      // Add a redirect to checkout parameter
      const redirectInput = document.createElement('input');
      redirectInput.type = 'hidden';
      redirectInput.name = 'checkout';
      redirectInput.value = '';
      form.appendChild(redirectInput);
      
      // Append the form to the body and submit it
      document.body.appendChild(form);
      form.submit();
    }
  }
  
  // Initialize on DOM content loaded
  document.addEventListener('DOMContentLoaded', () => {
    new OfferSelectedProducts();
  });
</script>

<script>
  document.addEventListener('DOMContentLoaded', function() {
    const productSelector = document.getElementById('offer-product-selector');
    const maxSelections = parseInt(productSelector.dataset.maxSelections) || 3;
    const selectionCountElement = document.getElementById('selection-count');
    const productCards = document.querySelectorAll('.offer-product-card');
    
    // Initialize selected products array
    window.selectedProducts = [];
    
    // Update selection count display
    function updateSelectionCount() {
      selectionCountElement.textContent = window.selectedProducts.length;
      
      // Dispatch custom event when selection changes
      const event = new CustomEvent('offer:selection-changed', {
        detail: {
          selectedProducts: window.selectedProducts,
          isComplete: window.selectedProducts.length === maxSelections
        }
      });
      document.dispatchEvent(event);
    }
    
    // Handle product selection
    productCards.forEach(card => {
      card.addEventListener('click', function() {
        const productId = this.dataset.productId;
        const cardInner = this.querySelector('.product-card-inner');
        const isSelected = cardInner.classList.contains('selected');
        
        // If already selected, remove from selection
        if (isSelected) {
          cardInner.classList.remove('selected');
          window.selectedProducts = window.selectedProducts.filter(p => p.id !== productId);
        } 
        // If not selected and haven't reached max, add to selection
        else if (window.selectedProducts.length < maxSelections) {
          cardInner.classList.add('selected');
          window.selectedProducts.push({
            id: productId,
            handle: this.dataset.productHandle,
            title: this.dataset.productTitle,
            price: this.dataset.productPrice,
            image: this.dataset.productImage,
            url: this.dataset.productUrl,
            variantId: this.dataset.variantId
          });
        }
        // If trying to select more than max, show message
        else {
          const existingAlert = document.querySelector('.max-selection-alert');
          if (!existingAlert) {
            const alert = document.createElement('div');
            alert.className = 'max-selection-alert';
            alert.textContent = `You can only select ${maxSelections} products. Please deselect one before adding another.`;
            alert.style.position = 'fixed';
            alert.style.top = '20px';
            alert.style.left = '50%';
            alert.style.transform = 'translateX(-50%)';
            alert.style.backgroundColor = '#ff4d4d';
            alert.style.color = 'white';
            alert.style.padding = '15px 20px';
            alert.style.borderRadius = '5px';
            alert.style.zIndex = '1000';
            alert.style.boxShadow = '0 5px 15px rgba(0,0,0,0.2)';
            document.body.appendChild(alert);
            
            setTimeout(() => {
              alert.remove();
            }, 3000);
          }
        }
        
        updateSelectionCount();
      });
    });
  });
</script>

<script>
  document.addEventListener('DOMContentLoaded', function() {
    // Set the date we're counting down to
    var countDownDate = new Date("{{ end_date }}").getTime();
    var timerId = "{{ timer_id }}";
    
    // Update the countdown every 1 second
    var x = setInterval(function() {
      // Get current date and time
      var now = new Date().getTime();
      
      // Find the distance between now and the countdown date
      var distance = countDownDate - now;
      
      // If the countdown is finished, display zeros
      if (distance < 0) {
        clearInterval(x);
        document.getElementById(timerId + "-days").innerHTML = "00";
        document.getElementById(timerId + "-hours").innerHTML = "00";
        document.getElementById(timerId + "-minutes").innerHTML = "00";
        document.getElementById(timerId + "-seconds").innerHTML = "00";
        return;
      }
      
      // Time calculations for days, hours, minutes and seconds
      var days = Math.floor(distance / (1000 * 60 * 60 * 24));
      var hours = Math.floor((distance % (1000 * 60 * 60 * 24)) / (1000 * 60 * 60));
      var minutes = Math.floor((distance % (1000 * 60 * 60)) / (1000 * 60));
      var seconds = Math.floor((distance % (1000 * 60)) / 1000);
      
      // Add leading zeros if needed
      days = days < 10 ? "0" + days : days;
      hours = hours < 10 ? "0" + hours : hours;
      minutes = minutes < 10 ? "0" + minutes : minutes;
      seconds = seconds < 10 ? "0" + seconds : seconds;
      
      // Display the result
      document.getElementById(timerId + "-days").innerHTML = days;
      document.getElementById(timerId + "-hours").innerHTML = hours;
      document.getElementById(timerId + "-minutes").innerHTML = minutes;
      document.getElementById(timerId + "-seconds").innerHTML = seconds;
    }, 1000);
  });
</script>
